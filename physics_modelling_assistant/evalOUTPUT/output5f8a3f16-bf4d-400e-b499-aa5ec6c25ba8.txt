\begin{align}
\gamma_{ij}(x,\rho)&= \gamma^{(0)}_{ij}(x)+ \gamma^{(1)}_{ij}(x)\rho +\gamma^{(2)}_{ij}(x)\rho^2 + \cdots\,.
\end{align}
